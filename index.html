grade_calculator.py 
marks = []

print("Student Grade Calculator")

for i in range(1, 6):
    mark = float(input(f"Enter mark {i}: "))
    marks.append(mark)

average = sum(marks) / len(marks)

print(f"\nAverage mark: {average:.2f}")

if average >= 90:
    grade = "A"
elif average >= 80:
    grade = "B"
elif average >= 70:
    grade = "C"
elif average >= 60:
    grade = "D"
else:
    grade = "F"

print(f"Grade: {grade}")
number_guessing.py 
import random

secret_number = random.randint(1, 100)
attempts = 0

print("Guess a number between 1 and 100")

while True:
    guess = int(input("Enter your guess: "))
    attempts += 1

    if guess < secret_number:
        print("Too low. Try again.")
    elif guess > secret_number:
        print("Too high. Try again.")
    else:
        print(f"Correct! You won in {attempts} attempts.")
        break
simple_calculator.py 
num1 = float(input("Enter first number: "))
operator = input("Enter operator (+, -, *, /): ")
num2 = float(input("Enter second number: "))

if operator == "+":
    result = num1 + num2
elif operator == "-":
    result = num1 - num2
elif operator == "*":
    result = num1 * num2
elif operator == "/" and num2 != 0:
    result = num1 / num2
else:
    result = "Invalid operation"

print("Result:", result)
calculator.c 
#include <stdio.h>

int main() {
    float first, second;
    char operator;

    printf("Enter first number: ");
    scanf("%f", &first);

    printf("Enter operator (+, -, *, /): ");
    scanf(" %c", &operator);

    printf("Enter second number: ");
    scanf("%f", &second);

    switch (operator) {
        case '+':
            printf("Result: %.2f\n", first + second);
            break;

        case '-':
            printf("Result: %.2f\n", first - second);
            break;

        case '*':
            printf("Result: %.2f\n", first * second);
            break;

        case '/':
            if (second != 0)
                printf("Result: %.2f\n", first / second);
            else
                printf("Cannot divide by zero.\n");
            break;

        default:
            printf("Invalid operator.\n");
    }

    return 0;
}
student_record.c 
#include <stdio.h>

struct Student {
    char name[50];
    int age;
    float mark;
};

int main() {
    struct Student student;

    printf("Enter student name: ");
    scanf(" %[^\n]", student.name);

    printf("Enter age: ");
    scanf("%d", &student.age);

    printf("Enter mark: ");
    scanf("%f", &student.mark);

    printf("\n--- Student Record ---\n");
    printf("Name: %s\n", student.name);
    printf("Age: %d\n", student.age);
    printf("Mark: %.2f\n", student.mark);

    return 0;
}
factorial.c 
#include <stdio.h>

int main() {
    int number;
    long long factorial = 1;

    printf("Enter a positive number: ");
    scanf("%d", &number);

    if (number < 0) {
        printf("Factorial is not defined for negative numbers.\n");
    } else {
        for (int i = 1; i <= number; i++) {
            factorial *= i;
        }

        printf("Factorial of %d = %lld\n",
               number, factorial);
    }

    return 0;
}
